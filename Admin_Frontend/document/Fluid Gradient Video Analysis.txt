2. 着色器架构设计（shaders.js）
此模块包含两个核心片段着色器：fluidSimulationFragmentShader 负责通过单纯形噪声和平流逻辑迭代流体的物理状态；displayFragmentShader 负责将物理状态通过余弦调色板映射为最终的人眼视觉色彩。






JavaScript
// shaders.js

// 通用顶点着色器：负责将二维几何体正确映射到屏幕空间
const vertexShader = `
   varying vec2 vUv;
   void main() {
       vUv = uv; // 捕获纹理坐标
       gl_Position = projectionMatrix * modelViewMatrix * vec4(position, 1.0);
   }
`;

// 流体模拟着色器（物理层）
const fluidSimulationFragmentShader = `
   uniform sampler2D tMap;
   uniform vec2 iResolution;
   uniform vec2 iMouse;
   uniform vec2 iVelocity;
   uniform float iTime;
   
   varying vec2 vUv;

   // Ashima Arts 三维单纯形噪声实现 (Simplex Noise 3D)
   vec3 mod289(vec3 x) { return x - floor(x * (1.0 / 289.0)) * 289.0; }
   vec4 mod289(vec4 x) { return x - floor(x * (1.0 / 289.0)) * 289.0; }
   vec4 permute(vec4 x) { return mod289(((x*34.0)+1.0)*x); }
   vec4 taylorInvSqrt(vec4 r) { return 1.79284291400159 - 0.85373472095314 * r; }

   float snoise(vec3 v) {
       const vec2  C = vec2(1.0/6.0, 1.0/3.0) ;
       const vec4  D = vec4(0.0, 0.5, 1.0, 2.0);

       vec3 i  = floor(v + dot(v, C.yyy) );
       vec3 x0 = v - i + dot(i, C.xxx) ;

       vec3 g = step(x0.yzx, x0.xyz);
       vec3 l = 1.0 - g;
       vec3 i1 = min( g.xyz, l.zxy );
       vec3 i2 = max( g.xyz, l.zxy );

       vec3 x1 = x0 - i1 + C.xxx;
       vec3 x2 = x0 - i2 + C.yyy;
       vec3 x3 = x0 - D.yyy;

       i = mod289(i); 
       vec4 p = permute( permute( permute( 
                  i.z + vec4(0.0, i1.z, i2.z, 1.0 ))
                + i.y + vec4(0.0, i1.y, i2.y, 1.0 )) 
                + i.x + vec4(0.0, i1.x, i2.x, 1.0 ));

       float n_ = 0.142857142857; 
       vec3  ns = n_ * D.wyz - D.xzx;

       vec4 j = p - 49.0 * floor(p * ns.z * ns.z);  

       vec4 x_ = floor(j * ns.z);
       vec4 y_ = floor(j - 7.0 * x_ );    

       vec4 x = x_ *ns.x + ns.yyyy;
       vec4 y = y_ *ns.x + ns.yyyy;
       vec4 h = 1.0 - abs(x) - abs(y);

       vec4 b0 = vec4( x.xy, y.xy );
       vec4 b1 = vec4( x.zw, y.zw );

       vec4 s0 = floor(b0)*2.0 + 1.0;
       vec4 s1 = floor(b1)*2.0 + 1.0;
       vec4 sh = -step(h, vec4(0.0));

       vec4 a0 = b0.xzyw + s0.xzyw*sh.xxyy ;
       vec4 a1 = b1.xzyw + s1.xzyw*sh.zzww ;

       vec3 p0 = vec3(a0.xy,h.x);
       vec3 p1 = vec3(a0.zw,h.y);
       vec3 p2 = vec3(a1.xy,h.z);
       vec3 p3 = vec3(a1.zw,h.w);

       vec4 norm = taylorInvSqrt(vec4(dot(p0,p0), dot(p1,p1), dot(p2, p2), dot(p3,p3)));
       p0 *= norm.x;
       p1 *= norm.y;
       p2 *= norm.z;
       p3 *= norm.w;

       vec4 m = max(0.5 - vec4(dot(x0,x0), dot(x1,x1), dot(x2,x2), dot(x3,x3)), 0.0);
       m = m * m;
       return 42.0 * dot( m*m, vec4( dot(p0,x0), dot(p1,x1), 
                                     dot(p2,x2), dot(p3,x3) ) );
   }

   void main() {
       // 读取上一帧的缓冲状态
       vec4 previousData = texture2D(tMap, vUv);
       
       // 域扭曲与平流：基于噪声计算底层流场速度
       float noiseScale = 3.0;
       float n1 = snoise(vec3(vUv * noiseScale, iTime * 0.1));
       float n2 = snoise(vec3(vUv * noiseScale + 10.0, iTime * 0.1));
       vec2 flowVelocity = vec2(n1, n2) * 0.005;

       // 根据流场速度向后偏移UV坐标，采样历史数据
       vec2 advectedUv = vUv - flowVelocity;
       vec4 advectedData = texture2D(tMap, advectedUv);

       // 鼠标交互外力场注入 (Mouse Splat)
       vec2 aspect = vec2(iResolution.x / iResolution.y, 1.0);
       vec2 mousePos = iMouse / iResolution;
       
       // 计算当前像素到鼠标位置的距离，考虑屏幕宽高比
       float dist = distance(vUv * aspect, mousePos * aspect);
       float splatRadius = 0.05;
       // 使用指数衰减生成柔和的外力注入点
       float splat = exp(-dist / splatRadius);
       
       // 根据鼠标移动速度施加动量
       vec2 force = splat * iVelocity * 0.5;
       
       // 综合平流数据与交互外力，并引入散度衰减防止能量爆炸
       vec2 finalData = advectedData.xy + force;
       finalData *= 0.99; // 粘度/衰减系数

       gl_FragColor = vec4(finalData, 0.0, 1.0);
   }
`;

// 色彩显示着色器（渲染层）
const displayFragmentShader = `
   uniform sampler2D tMap;
   varying vec2 vUv;

   // Inigo Quilez 余弦程序化调色板
   vec3 palette( in float t, in vec3 a, in vec3 b, in vec3 c, in vec3 d ) {
       return a + b*cos( 6.28318*(c*t+d) );
   }

   void main() {
       // 读取由物理层输出的流体模拟数据
       vec4 fluidData = texture2D(tMap, vUv);
       
       // 提取流体的运动强度（模长）作为色彩驱动参数
       float intensity = length(fluidData.xy) * 2.0;

       // 设定余弦调色板参数，打造蓝色/紫色系的现代美学渐变
       vec3 a = vec3(0.5, 0.5, 0.5);
       vec3 b = vec3(0.5, 0.5, 0.5);
       vec3 c = vec3(1.0, 1.0, 1.0);
       vec3 d = vec3(0.263, 0.416, 0.557);

       // 生成最终颜色
       vec3 color = palette(intensity, a, b, c, d);
       
       // 背景融合处理：当流体强度接近0时，渐隐为深色背景
       vec3 background = vec3(0.02);
       color = mix(background, color, min(intensity, 1.0));

       // 输出颜色值。Three.js的渲染管线会在必要时进行sRGB校正。
       gl_FragColor = vec4(color, 1.0);
   }
`;

3. 应用层逻辑与事件监听引擎（script.js）
最后，JavaScript层负责协调WebGL上下文、分配内存空间，并以60 FPS（或设备最高刷新率）执行乒乓交换策略。






JavaScript
// script.js

const canvas = document.getElementById('webgl-canvas');
let width = window.innerWidth;
let height = window.innerHeight;

// 1. 渲染器实例化与配置
const renderer = new THREE.WebGLRenderer({ canvas, antialias: false });
renderer.setSize(width, height);
// 严格控制像素比以保护移动端GPU填充率，避免过热
renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2)); 

// 2. 场景与相机拓扑
// 采用正交相机确保完美覆盖二维屏幕空间，忽略Z轴透视
const camera = new THREE.OrthographicCamera(width / -2, width / 2, height / 2, height / -2, 1, 1000);
camera.position.z = 1;
const scene = new THREE.Scene();

// 构建占满全屏的几何画板
const geometry = new THREE.PlaneGeometry(width, height);

// 3. 乒乓缓冲区(FBO)内存分配
// 采用 THREE.HalfFloatType 以支持亚像素级的高精度累积运算
const renderTargetOptions = {
   minFilter: THREE.LinearFilter,
   magFilter: THREE.LinearFilter,
   format: THREE.RGBAFormat,
   type: THREE.HalfFloatType 
};

let fluidTarget1 = new THREE.WebGLRenderTarget(width, height, renderTargetOptions);
let fluidTarget2 = new THREE.WebGLRenderTarget(width, height, renderTargetOptions);

// 4. 着色器材质注入
const fluidMaterial = new THREE.ShaderMaterial({
   vertexShader: vertexShader,
   fragmentShader: fluidSimulationFragmentShader,
   uniforms: {
       tMap: { value: null },
       iResolution: { value: new THREE.Vector2(width, height) },
       iMouse: { value: new THREE.Vector2(0, 0) },
       iVelocity: { value: new THREE.Vector2(0, 0) },
       iTime: { value: 0 }
   },
   depthWrite: false, // 禁用深度测试提升渲染效能
   depthTest: false
});

const displayMaterial = new THREE.ShaderMaterial({
   vertexShader: vertexShader,
   fragmentShader: displayFragmentShader,
   uniforms: {
       tMap: { value: null }
   }
});

const mesh = new THREE.Mesh(geometry, fluidMaterial);
scene.add(mesh);

// 5. 事件总线与交互物理量提取
let mouse = new THREE.Vector2(0, 0);
let lastMouse = new THREE.Vector2(0, 0);
let velocity = new THREE.Vector2(0, 0);

window.addEventListener('mousemove', (e) => {
   lastMouse.copy(mouse);
   
   // Y轴翻转以匹配WebGL原点位于左下角的纹理坐标系
   mouse.x = e.clientX;
   mouse.y = height - e.clientY;

   velocity.x = mouse.x - lastMouse.x;
   velocity.y = mouse.y - lastMouse.y;
   
   fluidMaterial.uniforms.iMouse.value.set(mouse.x, mouse.y);
   fluidMaterial.uniforms.iVelocity.value.set(velocity.x, velocity.y);
});

// 计算速度自然衰减（动量丧失）
function updateVelocity() {
   velocity.multiplyScalar(0.9); 
   fluidMaterial.uniforms.iVelocity.value.copy(velocity);
}

// 动态响应窗口重绘以避免视口拉伸
window.addEventListener("resize", () => {
   width = window.innerWidth;
   height = window.innerHeight;

   renderer.setSize(width, height);
   
   // 更新正交相机的截头锥体
   camera.left = width / -2;
   camera.right = width / 2;
   camera.top = height / 2;
   camera.bottom = height / -2;
   camera.updateProjectionMatrix();

   // 释放旧内存并重构几何体比例
   mesh.geometry.dispose();
   mesh.geometry = new THREE.PlaneGeometry(width, height);

   fluidMaterial.uniforms.iResolution.value.set(width, height);
   fluidTarget1.setSize(width, height);
   fluidTarget2.setSize(width, height);
});

// 6. 核心渲染循环（Ping-Pong Swap 机制）
const clock = new THREE.Clock();

function animate() {
   requestAnimationFrame(animate);
   
   const elapsedTime = clock.getElapsedTime();
   updateVelocity();
   fluidMaterial.uniforms.iTime.value = elapsedTime;

   // 通道A：将旧目标1数据通过着色器结算后写入目标2
   fluidMaterial.uniforms.tMap.value = fluidTarget1.texture;
   mesh.material = fluidMaterial;
   renderer.setRenderTarget(fluidTarget2);
   renderer.render(scene, camera);

   // 状态交换：目标2晋升为包含最新历史状态的目标1
   let temp = fluidTarget1;
   fluidTarget1 = fluidTarget2;
   fluidTarget2 = temp;

   // 通道B：读取包含最新状态的目标1，将视觉映射渲染至真实屏幕 (RenderTarget = null)
   displayMaterial.uniforms.tMap.value = fluidTarget1.texture;
   mesh.material = displayMaterial;
   renderer.setRenderTarget(null);
   renderer.render(scene, camera);
}

animate();

性能优化瓶颈与WebGPU的未来展望
虽然这种交互式流体渐变能够提供惊艳的视觉反馈，但其引入的硬件开销在专业开发环境（如对设备续航和温控极其敏感的移动端部署）中必须得到严格的分析和审视。
